cmake_minimum_required(VERSION 3.0.2)
project(logistics_gazebo_sim)
find_package(catkin REQUIRED COMPONENTS rospy geometry_msgs mavros_msgs gazebo_ros)
find_package(ompl REQUIRED)
find_package(gazebo REQUIRED)
catkin_python_setup()
catkin_package()
include_directories(${catkin_INCLUDE_DIRS} ${OMPL_INCLUDE_DIRS})
add_executable(ompl_3d_planner src/ompl_3d_planner.cpp)
target_link_libraries(ompl_3d_planner ${catkin_LIBRARIES} ${OMPL_LIBRARIES})
install(TARGETS ompl_3d_planner RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
add_executable(gazebo_mesh_triangles src/gazebo_mesh_triangles.cpp)
target_include_directories(gazebo_mesh_triangles PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_libraries(gazebo_mesh_triangles ${GAZEBO_LIBRARIES})
set_property(TARGET gazebo_mesh_triangles PROPERTY CXX_STANDARD 17)
install(TARGETS gazebo_mesh_triangles RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
add_executable(gazebo_world_ray_probe src/gazebo_world_ray_probe.cpp)
target_include_directories(gazebo_world_ray_probe PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_libraries(gazebo_world_ray_probe ${GAZEBO_LIBRARIES})
set_property(TARGET gazebo_world_ray_probe PROPERTY CXX_STANDARD 17)
install(TARGETS gazebo_world_ray_probe RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
catkin_install_python(PROGRAMS scripts/generate_worlds scripts/audit_world_assets scripts/generate_missions scripts/analyze_clearance scripts/analyze_track_reliability scripts/perception_acceptance_matrix scripts/perception_stale_trial scripts/energy_model_trial scripts/energy_return_trial scripts/energy_landing_trial scripts/track_continuity_trial scripts/safety_interlock_trial scripts/gazebo_marker_sync scripts/offboard_controller scripts/fleet_mission_player scripts/fleet_energy_monitor scripts/energy_return_advisor scripts/fleet_safety_monitor scripts/fleet_visualizer scripts/mission_recorder scripts/parcel_delivery_manager scripts/dynamic_obstacle_simulator scripts/dynamic_obstacle_sensor_bridge scripts/dynamic_target_tracker scripts/perception_visualizer scripts/lidar_cloud_aggregator scripts/lidar_dynamic_detector scripts/dynamic_risk_monitor scripts/dynamic_avoidance_planner scripts/benchmark_local_avoidance
  DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch config worlds models resources DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(FILES plugin.xml DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
